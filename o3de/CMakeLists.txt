cmake_minimum_required(VERSION 3.24)
project(unreal_unity_o3de_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(rust_engine_client STATIC core/RustEngineClient.cpp)
target_include_directories(rust_engine_client PUBLIC include core)
target_compile_options(rust_engine_client PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Werror>)
include(CTest)
if(BUILD_TESTING)
  add_executable(rust_engine_client_test tests/client_test.cpp tests/fake_rust_engine.cpp)
  target_link_libraries(rust_engine_client_test PRIVATE rust_engine_client)
  target_compile_options(rust_engine_client_test PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Werror>)
  add_test(NAME rust_engine_client_test COMMAND rust_engine_client_test)
endif()
